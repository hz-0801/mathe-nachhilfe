\begin{aufgabe}{Ich kann negative Zahlen addieren und subtrahieren. Rechne.}
\begin{teilezwei}
\tz $9 - 4 =$ \leerfeld & \tz $-2 + 6 =$ \leerfeld \\
\tz $1{,}5 - 4 =$ \leerfeld & \tz $3 - 8 =$ \leerfeld \\
\tz $-4 - 6 =$ \leerfeld & \tz $6 - 10 + 3 =$ \leerfeld \\
\end{teilezwei}
\end{aufgabe}

\begin{aufgabe}{Ich kann mit Vorzeichen multiplizieren und Punkt vor Strich beachten. Rechne.}

\tz $5 \cdot (-3) =$ \leerfeld & \tz $(-4) \cdot (-5) =$ \leerfeld \\
\tz $(-0{,}5) \cdot 8 =$ \leerfeld & \tz $(-8)^2 =$ \leerfeld \\
\tz $10 - 2 \cdot 7 =$ \leerfeld & \\
\end{teilezwei}
\end{aufgabe}

\begin{aufgabe}{Ich kann den Wert eines Terms berechnen. Setze $x = 3$ ein und berechne den Wert des Terms.}
\begin{teilezwei}
\tz $2x + 1 =$ \leerfeld & \tz $5x - 4 =$ \leerfeld \\
\end{teilezwei}
Setze jetzt $x = -2$ ein.
\begin{teilezwei}
\tz $3x + 4 =$ \leerfeld & \tz $x^2 + 1 =$ \leerfeld \\
\tz $7 - 2x =$ \leerfeld & \\
\end{teilezwei}
\end{aufgabe}

\begin{aufgabe}{Ich kann gleichartige Glieder zusammenfassen. Fasse zusammen.}
\begin{teilezwei}
\tz $4x + 3x =$ \leerfeld & \tz $9y - 5y =$ \leerfeld \\
\tz $2m + 5 + 6m =$ \leerfeld & \tz $z + 4z =$ \leerfeld \\
\tz $3n - 7n =$ \leerfeld & \tz $2x^2 + 5x + 3x^2 =$ \leerfeld \\
\tz $5m + 2n - 3m + 4n =$ \leerfeld & \\
\end{teilezwei}
\end{aufgabe}
