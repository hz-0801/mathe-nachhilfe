\begin{aufgabe}{Ich kann negative Zahlen addieren und subtrahieren.}
\anweisung{Rechne.}
\begin{teilezwei}
\tz $4 - 9 = $ \leerfeld & \tz $-3 - 6 = $ \leerfeld \\
\tz $-14 + 8 = $ \leerfeld & \tz $-5 - (-12) = $ \leerfeld \\
\tz $9 - 15 + 4 = $ \leerfeld & \\
\end{teilezwei}
\end{aufgabe}

\begin{aufgabe}{Ich kann Zahlen mit Vorzeichen malnehmen.}

\tz $(-12) \cdot (-3) = $ \leerfeld & \tz $(-1) \cdot (-9) = $ \leerfeld \\

\end{aufgabe}

\begin{aufgabe}{Ich kann Punkt vor Strich rechnen.}
\anweisung{Rechne.}
\begin{teilezwei}
\tz $2 + 3 \cdot 4 = $ \leerfeld & \tz $20 - 2 \cdot 5 = $ \leerfeld \\
\tz $7 - 4 \cdot 3 = $ \leerfeld & \tz $5 - 2 \cdot (-3) = $ \leerfeld \\
\tz $3 \cdot 2 - 4 \cdot 5 = $ \leerfeld & \\
\end{teilezwei}
\end{aufgabe}

\begin{aufgabe}{Ich kann den Wert eines Terms berechnen.}
\anweisung{Setze ein und berechne den Wert des Terms.}
\begin{teilezwei}
\tz $3x + 2$ für $x = 4$ \quad \leerfeld & \tz $5 - x$ für $x = 2$ \quad \leerfeld \\
\tz $2x - 7$ für $x = -3$ \quad \leerfeld & \tz $x^2 - 3$ für $x = -4$ \quad \leerfeld \\
\tz $4 - 2x$ für $x = -5$ \quad \leerfeld & \\
\end{teilezwei}
\end{aufgabe}

\begin{aufgabe}{Ich kann gleichartige Glieder zusammenfassen.}
\anweisung{Fasse zusammen.}
\begin{teilezwei}
\tz $4x + 5x = $ \leerfeld & \tz $9y - 3y = $ \leerfeld \\
\tz $2x + 7 + 5x = $ \leerfeld & \tz $x + 4x = $ \leerfeld \\
\tz $2y - 9y = $ \leerfeld & \tz $-3x + 5x = $ \leerfeld \\
\end{teilezwei}
\end{aufgabe}

\begin{aufgabe}{Ich kann Terme malnehmen.}
\anweisung{Multipliziere.}
\begin{teilezwei}
\tz $3 \cdot 5x = $ \leerfeld & \tz $4y \cdot 2 = $ \leerfeld \\
\tz $x \cdot x = $ \leerfeld & \tz $2x \cdot 4x = $ \leerfeld \\
\tz $(-3) \cdot 6x = $ \leerfeld & \tz $4x \cdot 5y = $ \leerfeld \\
\end{teilezwei}
\end{aufgabe}

\begin{aufgabe}{Ich finde den Fehler beim Zusammenfassen.}
\anweisung{Finde den Fehler, benenne ihn und korrigiere die Rechnung.}
\begin{teile}
\teil Ben fasst zusammen: \rechnung{3x^2 + 5x + 2x &= 10x^2}
\teil Lea fasst zusammen: \rechnung{4 + 3 \cdot x &= 7x}
\end{teile}
\end{aufgabe}

\begin{aufgabe}{Ich kann Terme mit $x$ und $x^2$ richtig zusammenfassen.}
\anweisung{Fasse zusammen.}
\begin{teilezwei}
\tz $5x^2 + 2x + 3x = $ \leerfeld & \tz $x^2 + 4x + 2x^2 = $ \leerfeld \\
\tz $6x - 2x^2 + x = $ \leerfeld & \tz $3x^2 + 7 - x^2 = $ \leerfeld \\
\tz $5 + 2 \cdot x + 3 = $ \leerfeld & \\
\end{teilezwei}
\end{aufgabe}

\begin{aufgabe}{Ich kann Quadratzahlen und Quadratwurzeln angeben.}
\anweisung{Berechne.}
\begin{teilezwei}
\tz $6^2 = $ \leerfeld & \tz $9^2 = $ \leerfeld \\
\tz $\sqrt{64} = $ \leerfeld & \tz $\sqrt{169} = $ \leerfeld \\
\tz $(-11)^2 = $ \leerfeld & \tz $(4x)^2 = $ \leerfeld \\
\anweisung{Schreibe als Quadrat.}
\tz $36 = (\;\leerfeld\;)^2$ & \tz $9x^2 = (\;\leerfeld\;)^2$ \\
\end{teilezwei}
\end{aufgabe}
